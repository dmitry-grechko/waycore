# App Manifest Reference

## Overview
Every app must include a `manifest.json` in its root directory.

## Required Fields
| Field | Type | Description |
|-------|------|-------------|
| id | string | Unique ID (reverse domain) |
| name | string | Display name (max 12 chars) |
| version | string | Semantic version (X.Y.Z) |
| icon | string | Emoji or icon name |
| entry.qml | string | Path to main QML file |

## Optional Fields
| Field | Type | Default | Description |
|-------|------|---------|-------------|
| description | string | null | Short description |
| category | enum | null | App category |
| entry.backend | string | null | Backend module path |
| permissions | array | [] | Required permissions |
| sensors | array | [] | Sensor requirements |
| database | object | null | Database configuration |
| tier | int | 2 | UI tier (1=home, 2=system) |
| enabled | bool | true | Whether app is enabled |

# Sensor Access Guide

## Available Sensors
| Type | Description | Data Format |
|------|-------------|-------------|
| magnetometer | Compass heading | {heading: float, declination: float} |
| gps | GPS location | {lat: float, lon: float, alt: float, accuracy: float} |
| barometer | Atmospheric pressure | {pressure: float, altitude: float} |
| temperature | Temperature | {celsius: float} |
| humidity | Relative humidity | {percent: float} |
| accelerometer | Motion | {x: float, y: float, z: float} |

## Declaring Sensor Requirements
```json
{
  "sensors": [
    {"type": "magnetometer", "required": true},
    {"type": "gps", "required": false}
  ]
}
```

## Accessing Sensor Data (QML)
```qml
import QtQuick 2.15

Item {
    // Check availability
    Component.onCompleted: {
        if (SensorAPI.isSensorAvailable("magnetometer")) {
            SensorAPI.subscribe("magnetometer")
        }
    }

    Component.onDestruction: {
        SensorAPI.unsubscribe("magnetometer")
    }

    Connections {
        target: SensorAPI
        function onSensorUpdated(type, data) {
            if (type === "magnetometer") {
                headingText.text = data.heading.toFixed(1) + "°"
            }
        }
    }
}
```

## Accessing Sensor Data (Python Backend)
```python
from device.apps.core.sensor_api import SensorAPI

class MyBackend:
    def __init__(self, sensor_api: SensorAPI):
        self._sensors = sensor_api

    def get_current_heading(self) -> float | None:
        data = self._sensors.get_latest("magnetometer")
        return data["heading"] if data else None
```

### 4. Database Patterns

```markdown
# App Database Guide

## Overview
Apps have two database access patterns:
1. **Shared Data (Read-only)**: Access core system data
2. **App Database (Read-write)**: Private app storage

## Declaring Database Needs
```json
{
  "database": {
    "version": 1,
    "sharedDataAccess": ["sensors.latest_readings"],
    "tables": [
      {
        "name": "my_data",
        "columns": [
          {"name": "id", "type": "TEXT", "primary": true},
          {"name": "value", "type": "TEXT"}
        ]
      }
    ]
  }
}
```

## Reading Shared Data
```qml
// Get latest GPS reading
var gps = DatabaseBridge.getLatestSensorReading("gps")
if (gps) {
    console.log("Location:", gps.latitude, gps.longitude)
}

// Get user setting
var units = DatabaseBridge.getUserSetting("units") || "metric"
```

## Using App Database
```qml
// Insert data
DatabaseBridge.execute(
    "INSERT INTO my_data (id, value) VALUES (?, ?)",
    [generateId(), "some value"]
)

// Query data
var results = DatabaseBridge.query("SELECT * FROM my_data ORDER BY id")
for (var row of results) {
    console.log(row.id, row.value)
}

### 6. Backend Patterns

```markdown
# App Backend Guide

## When to Use a Backend
- Complex data processing
- API integrations
- Background tasks
- Heavy computations

## Backend Structure
```python
# backend/service.py
from __future__ import annotations

from PySide6.QtCore import QObject, Signal, Slot


class MyAppBackend(QObject):
    """Backend service for My App."""

    dataChanged = Signal()

    def __init__(self, parent: QObject | None = None) -> None:
        super().__init__(parent)

    @Slot(str, result=str)
    def processData(self, input: str) -> str:
        # Process and return
        return input.upper()


def create_bridge() -> QObject:
    """Factory function called by the app loader."""
    return MyAppBackend()
```

## Accessing Core Services
```python
from device.apps.core.services import get_core_services

services = get_core_services()
sensor_api = services.get("sensor_api")
database = services.get("database")
```

## QML Integration
The backend is automatically registered as a context property.
```qml
// In QML
var result = MyAppBackend.processData("hello")
// result = "HELLO"
```

## App Scaffolding Script

```python
#!/usr/bin/env python3
"""
Create a new Waycore app from template.

Usage: python scripts/create-app.py my-app-name
"""

import argparse
import json
from pathlib import Path


def create_app(name: str, with_backend: bool = False) -> None:
    """Create a new app directory with template files."""

    # Convert name to ID format
    app_id = f"com.waycore.{name.replace('-', '')}"
    app_dir = Path(f"device/apps/{name}")

    if app_dir.exists():
        print(f"Error: {app_dir} already exists")
        return

    # Create directory structure
    app_dir.mkdir(parents=True)
    (app_dir / "qml").mkdir()
    if with_backend:
        (app_dir / "backend").mkdir()

    # Create manifest
    manifest = {
        "id": app_id,
        "name": name.replace("-", " ").title()[:12],
        "version": "1.0.0",
        "description": f"A Waycore app",
        "icon": "📱",
        "category": "tools",
        "entry": {
            "qml": "qml/Main.qml",
        },
        "tier": 2,
        "enabled": True,
    }

    if with_backend:
        manifest["entry"]["backend"] = "backend/service.py"

    with open(app_dir / "manifest.json", "w") as f:
        json.dump(manifest, f, indent=2)

    # Create main QML
    qml_content = '''import QtQuick 2.15
import "../../core/qml" as Core
import "../../core/qml/components" as UI

Rectangle {
    id: root
    color: Core.Theme.background

    signal closeRequested()

    property string appId: "%s"
    property string appTitle: "%s"

    UI.AppBar {
        id: appBar
        anchors.top: parent.top
        anchors.left: parent.left
        anchors.right: parent.right
        title: root.appTitle
        showBackButton: true
        onBackClicked: root.closeRequested()
    }

    // App content here
    Text {
        anchors.centerIn: parent
        text: "Hello from " + root.appTitle
        color: Core.Theme.textPrimary
        font.pixelSize: Core.Theme.h2Size
    }
}
''' % (app_id, manifest["name"])

    with open(app_dir / "qml" / "Main.qml", "w") as f:
        f.write(qml_content)

    # Create backend if requested
    if with_backend:
        backend_content = '''"""Backend service for %s."""
from __future__ import annotations

from PySide6.QtCore import QObject, Signal, Slot


class AppBackend(QObject):
    """Backend service."""

    def __init__(self, parent: QObject | None = None) -> None:
        super().__init__(parent)

    @Slot(result=str)
    def hello(self) -> str:
        return "Hello from backend!"


def create_bridge() -> QObject:
    """Factory function called by the app loader."""
    return AppBackend()
''' % manifest["name"]

        with open(app_dir / "backend" / "__init__.py", "w") as f:
            f.write("")
        with open(app_dir / "backend" / "service.py", "w") as f:
            f.write(backend_content)

    # Create README
    readme = f"""# {manifest["name"]}

A Waycore app.

## Development

1. Edit `qml/Main.qml` for the UI
{"2. Edit `backend/service.py` for backend logic" if with_backend else ""}

## Testing

Run the UI app and navigate to your app from the system hub.
"""

    with open(app_dir / "README.md", "w") as f:
        f.write(readme)

    print(f"✅ Created app: {app_dir}")
    print(f"   ID: {app_id}")
    print(f"   Name: {manifest['name']}")
    if with_backend:
        print(f"   Backend: enabled")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Create a new Waycore app")
    parser.add_argument("name", help="App name (lowercase, hyphens allowed)")
    parser.add_argument("--backend", action="store_true", help="Include backend module")
    args = parser.parse_args()

    create_app(args.name, args.backend)
```
