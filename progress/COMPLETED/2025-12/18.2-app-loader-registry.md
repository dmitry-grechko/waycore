## Description

Implement the core app loader and registry system that discovers, validates, and registers apps from the filesystem. This is the runtime system that makes the modular architecture work.

## Context

The app loader scans the `device/apps/` directory for valid app packages (directories containing `manifest.json`), validates them, and makes them available to both the QML frontend and Python backend.

## Implementation

### App Loader

```python
from __future__ import annotations

import json
from pathlib import Path
from typing import Optional

from device.libs.schemas.app_manifest import AppManifest


class AppLoader:
    """Discovers and loads apps from the filesystem."""

    def __init__(self, apps_dir: Path) -> None:
        self.apps_dir = apps_dir

    def discover_apps(self) -> list[AppManifest]:
        """Scan apps directory for valid app packages."""
        apps: list[AppManifest] = []

        for app_dir in self.apps_dir.iterdir():
            if not app_dir.is_dir():
                continue
            if app_dir.name.startswith(("_", ".")):
                continue
            if app_dir.name in ("core", "ui"):  # Skip special directories
                continue

            manifest_path = app_dir / "manifest.json"
            if manifest_path.exists():
                try:
                    manifest = self._load_manifest(manifest_path, app_dir.name)
                    if manifest.enabled:
                        apps.append(manifest)
                except Exception as e:
                    print(f"Warning: Failed to load app {app_dir.name}: {e}")

        return apps

    def _load_manifest(self, path: Path, app_name: str) -> AppManifest:
        """Load and validate a manifest file."""
        with open(path) as f:
            data = json.load(f)

        # Store the app directory for resolving paths
        manifest = AppManifest(**data)
        manifest._app_dir = path.parent  # Internal reference
        return manifest
```

### App Registry

```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Optional

from device.libs.schemas.app_manifest import AppManifest


@dataclass
class LoadedApp:
    """A loaded app with resolved paths."""
    manifest: AppManifest
    app_dir: Path
    qml_entry: Path
    backend_module: Optional[object] = None


class AppRegistry:
    """Runtime registry of loaded apps."""

    def __init__(self) -> None:
        self._apps: dict[str, LoadedApp] = {}

    def register(self, app: LoadedApp) -> None:
        """Register a loaded app."""
        self._apps[app.manifest.id] = app

    def get(self, app_id: str) -> Optional[LoadedApp]:
        """Get a loaded app by ID."""
        return self._apps.get(app_id)

    def get_tier1_apps(self) -> list[AppManifest]:
        """Get apps for the home screen (tier 1)."""
        apps = [a.manifest for a in self._apps.values() if a.manifest.tier == 1]
        # Sort by homePosition if set, otherwise alphabetically
        return sorted(apps, key=lambda a: (a.home_position or 999, a.name))

    def get_tier2_apps(self) -> list[AppManifest]:
        """Get apps for the system hub (tier 2)."""
        apps = [a.manifest for a in self._apps.values() if a.manifest.tier == 2]
        return sorted(apps, key=lambda a: (a.category or "", a.name))

    def get_all_apps(self) -> list[AppManifest]:
        """Get all registered apps."""
        return [a.manifest for a in self._apps.values()]
```

### QML Bridge

```python
from __future__ import annotations

from PySide6.QtCore import Property, QObject, Signal, Slot

from .registry import AppRegistry


class AppBridge(QObject):
    """QML bridge for accessing app registry."""

    appsChanged = Signal()

    def __init__(self, registry: AppRegistry, parent: QObject | None = None) -> None:
        super().__init__(parent)
        self._registry = registry

    @Property(list, notify=appsChanged)  # type: ignore[arg-type]
    def tier1Apps(self) -> list[dict]:
        """Apps for home screen."""
        return [self._manifest_to_dict(m) for m in self._registry.get_tier1_apps()]

    @Property(list, notify=appsChanged)  # type: ignore[arg-type]
    def tier2Apps(self) -> list[dict]:
        """Apps for system hub."""
        return [self._manifest_to_dict(m) for m in self._registry.get_tier2_apps()]

    @Slot(str, result=str)  # type: ignore[arg-type]
    def getAppEntry(self, app_id: str) -> str:
        """Get QML entry path for an app."""
        app = self._registry.get(app_id)
        if app:
            return str(app.qml_entry)
        return ""

    @Slot(str, result=bool)  # type: ignore[arg-type]
    def isAppEnabled(self, app_id: str) -> bool:
        """Check if an app is enabled."""
        app = self._registry.get(app_id)
        return app is not None and app.manifest.enabled

    def _manifest_to_dict(self, m: AppManifest) -> dict:
        """Convert manifest to QML-friendly dict."""
        return {
            "id": m.id,
            "name": m.name,
            "icon": m.icon,
            "category": m.category.value if m.category else None,
            "tier": m.tier,
        }
```
