# Task: Recursive Factory Reset System

**Priority**: High

## Description

Implement a comprehensive factory reset system that works independently of app structure. Currently, factory reset manually clears specific DB tables and storage for each service. With the modular app ecosystem, this approach doesn't scale - each new app would require manual reset code.

The new system should:
1. Automatically discover and clear all databases
2. Automatically clear all app storage
3. Reset system settings to defaults
4. Work without knowledge of individual app schemas

## Current State (Problems)

1. **Manual reset per service**: Each service has custom reset code
2. **Schema-dependent**: Reset logic knows about specific tables
3. **Not scalable**: New apps require new reset code
4. **Incomplete**: Easy to miss new storage locations
5. **No app lifecycle hooks**: Apps can't prepare for reset

## Target State (Goals)

1. **Automatic discovery**: Find all databases and storage directories
2. **Schema-agnostic**: Delete entire databases, not specific tables
3. **Scalable**: Works with any number of apps automatically
4. **Complete**: Clears ALL data directories
5. **App notification**: Apps can respond to reset (cleanup hooks)
6. **Safe defaults**: System reinitializes with clean defaults

## Implementation

### Factory Reset Manager

```python
from __future__ import annotations

import os
import shutil
from pathlib import Path
from typing import Callable

from device.apps.core.registry import AppRegistry


class FactoryResetManager:
    """
    Manages complete factory reset of the system.

    This is schema-agnostic - it discovers and deletes all data
    without needing to know the structure of individual databases.
    """

    def __init__(
        self,
        data_dir: Path,
        storage_dir: Path,
        app_registry: AppRegistry,
    ) -> None:
        self._data_dir = data_dir
        self._storage_dir = storage_dir
        self._app_registry = app_registry
        self._pre_reset_hooks: list[Callable[[], None]] = []
        self._post_reset_hooks: list[Callable[[], None]] = []

    def register_pre_reset_hook(self, hook: Callable[[], None]) -> None:
        """Register a callback to run before reset."""
        self._pre_reset_hooks.append(hook)

    def register_post_reset_hook(self, hook: Callable[[], None]) -> None:
        """Register a callback to run after reset."""
        self._post_reset_hooks.append(hook)

    def reset_all(self, keep_system_logs: bool = False) -> ResetResult:
        """
        Perform complete factory reset.

        Args:
            keep_system_logs: If True, preserve system logs for debugging

        Returns:
            ResetResult with details of what was cleared
        """
        result = ResetResult()

        try:
            # 1. Notify apps/services of impending reset
            self._run_pre_reset_hooks()

            # 2. Clear all databases
            result.databases_cleared = self._clear_all_databases()

            # 3. Clear all app storage
            result.storage_cleared = self._clear_all_storage(keep_system_logs)

            # 4. Reset system defaults
            result.defaults_restored = self._restore_defaults()

            # 5. Notify apps/services reset is complete
            self._run_post_reset_hooks()

            result.success = True

        except Exception as e:
            result.success = False
            result.error = str(e)

        return result

    def _clear_all_databases(self) -> list[str]:
        """
        Find and delete ALL SQLite databases.

        This is intentionally schema-agnostic - we delete entire
        database files rather than specific tables.
        """
        cleared: list[str] = []

        # Core databases in /app/data/
        for db_file in self._data_dir.glob("*.sqlite3"):
            db_file.unlink()
            cleared.append(str(db_file.name))

        # App-specific databases in /app/data/apps/
        apps_db_dir = self._data_dir / "apps"
        if apps_db_dir.exists():
            for db_file in apps_db_dir.glob("*.sqlite3"):
                db_file.unlink()
                cleared.append(f"apps/{db_file.name}")

        # Also check for any .db files (alternative extension)
        for db_file in self._data_dir.rglob("*.db"):
            db_file.unlink()
            cleared.append(str(db_file.relative_to(self._data_dir)))

        return cleared

    def _clear_all_storage(self, keep_logs: bool = False) -> list[str]:
        """
        Clear all storage directories.

        Storage includes:
        - /app/storage/photos/
        - /app/storage/cache/
        - /app/storage/downloads/
        - /app/storage/apps/{app_id}/
        """
        cleared: list[str] = []

        if not self._storage_dir.exists():
            return cleared

        for item in self._storage_dir.iterdir():
            # Optionally keep logs
            if keep_logs and item.name == "logs":
                continue

            if item.is_dir():
                shutil.rmtree(item)
                cleared.append(item.name + "/")
            else:
                item.unlink()
                cleared.append(item.name)

        # Recreate essential directories
        self._ensure_storage_structure()

        return cleared

    def _ensure_storage_structure(self) -> None:
        """Recreate essential storage directories."""
        essential_dirs = [
            self._storage_dir / "photos",
            self._storage_dir / "cache",
            self._storage_dir / "downloads",
            self._storage_dir / "apps",
            self._storage_dir / "logs",
        ]
        for dir_path in essential_dirs:
            dir_path.mkdir(parents=True, exist_ok=True)

    def _restore_defaults(self) -> dict[str, any]:
        """
        Restore system to default configuration.

        This creates fresh databases with default values.
        """
        from device.apps.core.reset.defaults import SystemDefaults

        defaults = SystemDefaults(self._data_dir)
        return defaults.restore()

    def _run_pre_reset_hooks(self) -> None:
        """Run all pre-reset hooks."""
        for hook in self._pre_reset_hooks:
            try:
                hook()
            except Exception as e:
                print(f"Pre-reset hook failed: {e}")

    def _run_post_reset_hooks(self) -> None:
        """Run all post-reset hooks."""
        for hook in self._post_reset_hooks:
            try:
                hook()
            except Exception as e:
                print(f"Post-reset hook failed: {e}")


@dataclass
class ResetResult:
    """Result of a factory reset operation."""
    success: bool = False
    databases_cleared: list[str] = field(default_factory=list)
    storage_cleared: list[str] = field(default_factory=list)
    defaults_restored: dict[str, any] = field(default_factory=dict)
    error: str | None = None
```

### Default Values Restoration

```python
from __future__ import annotations

import sqlite3
from pathlib import Path


class SystemDefaults:
    """
    Restores system to default configuration.

    Creates fresh databases with essential default values.
    """

    def __init__(self, data_dir: Path) -> None:
        self._data_dir = data_dir

    def restore(self) -> dict[str, any]:
        """Restore all defaults and return summary."""
        results = {}

        # Create general.sqlite3 with defaults
        results["general"] = self._restore_general_database()

        # Create settings with defaults
        results["settings"] = self._restore_settings()

        return results

    def _restore_general_database(self) -> dict:
        """Create general.sqlite3 with default schema."""
        db_path = self._data_dir / "general.sqlite3"
        conn = sqlite3.connect(str(db_path))

        try:
            # Create essential tables
            conn.executescript("""
                CREATE TABLE IF NOT EXISTS sensor_readings (
                    id TEXT PRIMARY KEY,
                    sensor_type TEXT NOT NULL,
                    value TEXT NOT NULL,
                    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
                );

                CREATE TABLE IF NOT EXISTS locations (
                    id TEXT PRIMARY KEY,
                    name TEXT,
                    latitude REAL,
                    longitude REAL,
                    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
                );

                CREATE TABLE IF NOT EXISTS user_settings (
                    key TEXT PRIMARY KEY,
                    value TEXT
                );

                CREATE INDEX IF NOT EXISTS idx_sensor_type
                    ON sensor_readings(sensor_type);
                CREATE INDEX IF NOT EXISTS idx_sensor_timestamp
                    ON sensor_readings(timestamp);
            """)

            # Insert default settings
            default_settings = [
                ("units", "metric"),
                ("theme", "dark"),
                ("language", "en"),
                ("timezone", "UTC"),
                ("first_run", "true"),
            ]
            conn.executemany(
                "INSERT OR REPLACE INTO user_settings (key, value) VALUES (?, ?)",
                default_settings
            )
            conn.commit()

            return {"tables": 3, "default_settings": len(default_settings)}

        finally:
            conn.close()

    def _restore_settings(self) -> dict:
        """Restore system settings to defaults."""
        # This creates the core settings that the system needs
        return {
            "units": "metric",
            "theme": "dark",
            "language": "en",
            "first_run": True,
        }
```

### Reset API Endpoint

```python
from fastapi import APIRouter, HTTPException

router = APIRouter(prefix="/api/system", tags=["system"])


@router.post("/factory-reset")
async def factory_reset(
    confirm: bool = False,
    keep_logs: bool = False,
) -> dict:
    """
    Perform factory reset.

    This clears ALL data and restores system defaults.
    Requires explicit confirmation.
    """
    if not confirm:
        raise HTTPException(
            status_code=400,
            detail="Factory reset requires confirm=true parameter"
        )

    from device.apps.core.reset.factory_reset import FactoryResetManager

    manager = get_reset_manager()  # Singleton
    result = manager.reset_all(keep_system_logs=keep_logs)

    if not result.success:
        raise HTTPException(status_code=500, detail=result.error)

    return {
        "success": True,
        "databases_cleared": result.databases_cleared,
        "storage_cleared": result.storage_cleared,
        "message": "Factory reset complete. Please restart the device."
    }
```
