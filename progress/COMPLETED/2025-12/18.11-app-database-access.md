# Task: App Database Access Patterns

**Phase**: Phase 18 - Modular App Ecosystem
**Task ID**: 18.11
**Status**: COMPLETED
**Started**: 2025-12-29
**Completed**: 2025-12-29
**Priority**: High

## Description

Define and implement the database access patterns for modular apps. Apps need two capabilities:
1. **Read shared data**: Access well-defined tables in the core databases (sensors, settings, etc.)
2. **App-specific storage**: Create and manage their own SQLite database with migrations

## Context

Currently, database access is handled by service-specific classes (`GeneralDatabase`, `MeshDatabase`, `AIDatabase`). For the modular app ecosystem, we need:
- Clear documentation of shared database schemas apps can read
- A pattern for apps to create their own databases
- Migration system for app database schema changes
- Isolation to prevent apps from corrupting core data

## Shared Data Reader

Apps can read (but not write) specific tables from core databases:

```python
from __future__ import annotations

import sqlite3
from pathlib import Path
from typing import Any


class SharedDataReader:
    """
    Read-only access to shared/core database tables.

    Apps can access specific tables that are documented and stable.
    Write access is NOT provided - apps must use their own database.
    """

    # Allowed tables apps can read (explicitly whitelisted)
    ALLOWED_TABLES = {
        "general.sqlite3": {
            "sensor_readings": ["id", "sensor_type", "value", "timestamp"],
            "locations": ["id", "name", "latitude", "longitude", "timestamp"],
            "user_settings": ["key", "value"],
        },
        "mesh.sqlite3": {
            "nodes": ["node_id", "short_name", "long_name", "last_seen"],
            "messages": ["id", "from_node", "to_node", "text", "timestamp"],
        },
    }

    def __init__(self, data_dir: Path) -> None:
        self._data_dir = data_dir

    def query(
        self,
        database: str,
        table: str,
        columns: list[str] | None = None,
        where: str | None = None,
        params: tuple[Any, ...] = (),
        limit: int | None = None,
    ) -> list[dict[str, Any]]:
        """
        Execute a read-only query on a shared table.

        Args:
            database: Database name (e.g., "general.sqlite3")
            table: Table name (must be in ALLOWED_TABLES)
            columns: Columns to select (None = allowed columns only)
            where: WHERE clause (without WHERE keyword)
            params: Parameters for WHERE clause
            limit: Maximum rows to return

        Returns:
            List of row dictionaries

        Raises:
            PermissionError: If table/column access is not allowed
        """
        # Validate access
        if database not in self.ALLOWED_TABLES:
            raise PermissionError(f"Access to database '{database}' not allowed")

        allowed_tables = self.ALLOWED_TABLES[database]
        if table not in allowed_tables:
            raise PermissionError(f"Access to table '{table}' not allowed")

        allowed_cols = allowed_tables[table]
        if columns:
            for col in columns:
                if col not in allowed_cols:
                    raise PermissionError(f"Access to column '{col}' not allowed")
        else:
            columns = allowed_cols

        # Build query
        col_str = ", ".join(columns)
        sql = f"SELECT {col_str} FROM {table}"
        if where:
            sql += f" WHERE {where}"
        if limit:
            sql += f" LIMIT {limit}"

        # Execute read-only
        db_path = self._data_dir / database
        conn = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
        conn.row_factory = sqlite3.Row
        try:
            cursor = conn.execute(sql, params)
            return [dict(row) for row in cursor.fetchall()]
        finally:
            conn.close()

    def get_latest_sensor_reading(self, sensor_type: str) -> dict[str, Any] | None:
        """Convenience method: Get latest reading for a sensor type."""
        results = self.query(
            "general.sqlite3",
            "sensor_readings",
            where="sensor_type = ?",
            params=(sensor_type,),
            limit=1,
        )
        return results[0] if results else None

    def get_user_setting(self, key: str) -> Any:
        """Convenience method: Get a user setting value."""
        results = self.query(
            "general.sqlite3",
            "user_settings",
            columns=["value"],
            where="key = ?",
            params=(key,),
        )
        return results[0]["value"] if results else None
```

## App Database Manager

Each app gets its own SQLite database file:

```python
from __future__ import annotations

import sqlite3
from pathlib import Path
from typing import Any

from device.libs.schemas.app_manifest import AppManifest, DatabaseConfig


class AppDatabase:
    """
    Manages an app's private SQLite database.

    Each app gets: /app/data/apps/{app_id}.sqlite3
    """

    def __init__(self, app_id: str, data_dir: Path, config: DatabaseConfig) -> None:
        self._app_id = app_id
        self._config = config
        self._db_path = data_dir / "apps" / f"{app_id}.sqlite3"
        self._db_path.parent.mkdir(parents=True, exist_ok=True)
        self._conn: sqlite3.Connection | None = None

    def initialize(self) -> None:
        """Initialize database and run migrations."""
        self._conn = sqlite3.connect(str(self._db_path))
        self._conn.row_factory = sqlite3.Row

        # Create schema version table
        self._conn.execute("""
            CREATE TABLE IF NOT EXISTS _schema_version (
                version INTEGER PRIMARY KEY,
                applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)

        # Get current version
        cursor = self._conn.execute(
            "SELECT MAX(version) as version FROM _schema_version"
        )
        row = cursor.fetchone()
        current_version = row["version"] if row["version"] else 0

        # Apply migrations
        if current_version < self._config.version:
            self._run_migrations(current_version, self._config.version)

    def _run_migrations(self, from_version: int, to_version: int) -> None:
        """Run migrations from current to target version."""
        if from_version == 0:
            # Initial schema creation
            self._create_initial_schema()
            from_version = 1

        # Run incremental migrations
        for migration in self._config.migrations:
            if from_version < migration.version <= to_version:
                print(f"Running migration v{migration.version} for {self._app_id}")
                for statement in migration.up.split(";"):
                    statement = statement.strip()
                    if statement:
                        self._conn.execute(statement)
                self._conn.execute(
                    "INSERT INTO _schema_version (version) VALUES (?)",
                    (migration.version,)
                )

        self._conn.commit()

    def _create_initial_schema(self) -> None:
        """Create tables from manifest definition."""
        for table in self._config.tables:
            columns = []
            for col in table.columns:
                col_def = f"{col.name} {col.type}"
                if col.primary:
                    col_def += " PRIMARY KEY"
                if not col.nullable:
                    col_def += " NOT NULL"
                if col.default:
                    col_def += f" DEFAULT {col.default}"
                columns.append(col_def)

            sql = f"CREATE TABLE IF NOT EXISTS {table.name} ({', '.join(columns)})"
            self._conn.execute(sql)

            # Create indexes
            for idx in table.indexes:
                idx_cols = ", ".join(idx.columns)
                self._conn.execute(
                    f"CREATE INDEX IF NOT EXISTS {idx.name} ON {table.name} ({idx_cols})"
                )

        self._conn.execute("INSERT INTO _schema_version (version) VALUES (1)")
        self._conn.commit()

    def execute(self, sql: str, params: tuple[Any, ...] = ()) -> sqlite3.Cursor:
        """Execute a SQL statement."""
        return self._conn.execute(sql, params)

    def executemany(self, sql: str, params_list: list[tuple[Any, ...]]) -> sqlite3.Cursor:
        """Execute a SQL statement with multiple parameter sets."""
        return self._conn.executemany(sql, params_list)

    def fetchone(self, sql: str, params: tuple[Any, ...] = ()) -> dict[str, Any] | None:
        """Execute and fetch one row."""
        cursor = self._conn.execute(sql, params)
        row = cursor.fetchone()
        return dict(row) if row else None

    def fetchall(self, sql: str, params: tuple[Any, ...] = ()) -> list[dict[str, Any]]:
        """Execute and fetch all rows."""
        cursor = self._conn.execute(sql, params)
        return [dict(row) for row in cursor.fetchall()]

    def commit(self) -> None:
        """Commit the current transaction."""
        self._conn.commit()

    def close(self) -> None:
        """Close the database connection."""
        if self._conn:
            self._conn.close()
            self._conn = None
```

## Database Bridge for QML

```python
from __future__ import annotations

from PySide6.QtCore import Property, QObject, Signal, Slot

from .app_database import AppDatabase
from .shared_reader import SharedDataReader


class DatabaseBridge(QObject):
    """QML bridge for app database access."""

    errorOccurred = Signal(str)

    def __init__(
        self,
        app_db: AppDatabase,
        shared_reader: SharedDataReader,
        parent: QObject | None = None,
    ) -> None:
        super().__init__(parent)
        self._app_db = app_db
        self._shared = shared_reader

    # === App Database Methods ===

    @Slot(str, result="QVariant")  # type: ignore[arg-type]
    def query(self, sql: str) -> list[dict]:
        """Execute a query on the app's database."""
        try:
            return self._app_db.fetchall(sql)
        except Exception as e:
            self.errorOccurred.emit(str(e))
            return []

    @Slot(str, "QVariant", result=bool)  # type: ignore[arg-type]
    def execute(self, sql: str, params: list) -> bool:
        """Execute a statement on the app's database."""
        try:
            self._app_db.execute(sql, tuple(params or []))
            self._app_db.commit()
            return True
        except Exception as e:
            self.errorOccurred.emit(str(e))
            return False

    # === Shared Data Methods ===

    @Slot(str, result="QVariant")  # type: ignore[arg-type]
    def getLatestSensorReading(self, sensor_type: str) -> dict | None:
        """Get the latest sensor reading from shared data."""
        try:
            return self._shared.get_latest_sensor_reading(sensor_type)
        except Exception as e:
            self.errorOccurred.emit(str(e))
            return None

    @Slot(str, result="QVariant")  # type: ignore[arg-type]
    def getUserSetting(self, key: str) -> object:
        """Get a user setting from shared data."""
        try:
            return self._shared.get_user_setting(key)
        except Exception as e:
            self.errorOccurred.emit(str(e))
            return None
```
