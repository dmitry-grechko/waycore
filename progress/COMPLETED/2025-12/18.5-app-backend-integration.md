## Description

Define and implement the backend integration patterns for modular apps. This includes:
1. How apps access core services (sensors, storage, messaging)
2. How apps register their own backend modules
3. API patterns for app-specific endpoints
4. Sensor data access patterns

## Context

Apps need to interact with the backend for:
- **Sensors**: Reading magnetometer, GPS, barometer, etc.
- **Storage**: Persisting app data (notes, settings)
- **Messaging**: MQTT pub/sub for real-time updates
- **Services**: Calling other services (camera, mesh, AI)

Currently, bridges (SensorBridge, NotesBridge, etc.) are created in `main.py`. We need a pattern where:
1. Core bridges are always available
2. Apps can register additional bridges if needed
3. Apps can access sensors through a unified API

## Core Services Registry

```python
from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Callable

from PySide6.QtCore import QObject


@dataclass
class ServiceDefinition:
    """Definition of a core service."""
    name: str
    factory: Callable[[], QObject]
    singleton: bool = True


class CoreServices:
    """Registry of core services available to apps."""

    def __init__(self) -> None:
        self._services: dict[str, ServiceDefinition] = {}
        self._instances: dict[str, QObject] = {}

    def register(
        self,
        name: str,
        factory: Callable[[], QObject],
        singleton: bool = True
    ) -> None:
        """Register a core service."""
        self._services[name] = ServiceDefinition(
            name=name,
            factory=factory,
            singleton=singleton
        )

    def get(self, name: str) -> QObject | None:
        """Get a service instance."""
        if name not in self._services:
            return None

        service = self._services[name]

        if service.singleton:
            if name not in self._instances:
                self._instances[name] = service.factory()
            return self._instances[name]
        else:
            return service.factory()

    def get_all_names(self) -> list[str]:
        """Get all registered service names."""
        return list(self._services.keys())


# Global instance
_core_services = CoreServices()


def get_core_services() -> CoreServices:
    """Get the global core services registry."""
    return _core_services
```

## Sensor Access API

```python
from __future__ import annotations

from typing import Any, Callable

from PySide6.QtCore import Property, QObject, Signal, Slot


class SensorAPI(QObject):
    """
    Unified API for apps to access sensor data.

    Apps request sensors by type. The API handles:
    - Checking if sensor is available
    - Subscribing to sensor updates
    - Providing latest values
    - Handling sensor errors
    """

    # Signals for sensor updates
    sensorUpdated = Signal(str, object)  # (sensor_type, data)
    sensorError = Signal(str, str)       # (sensor_type, error_message)
    sensorAvailabilityChanged = Signal(str, bool)  # (sensor_type, available)

    def __init__(self, sensor_bridge: QObject, parent: QObject | None = None) -> None:
        super().__init__(parent)
        self._sensor_bridge = sensor_bridge
        self._subscriptions: dict[str, int] = {}  # sensor_type -> refcount
        self._latest_values: dict[str, Any] = {}

    @Slot(str, result=bool)  # type: ignore[arg-type]
    def isSensorAvailable(self, sensor_type: str) -> bool:
        """Check if a sensor type is available."""
        # Map to SensorBridge methods
        available_sensors = {
            "magnetometer": True,  # Mock always available
            "gps": True,
            "barometer": True,
            "temperature": True,
            "humidity": True,
        }
        return available_sensors.get(sensor_type, False)

    @Slot(str)  # type: ignore[arg-type]
    def subscribe(self, sensor_type: str) -> None:
        """Subscribe to sensor updates."""
        if sensor_type in self._subscriptions:
            self._subscriptions[sensor_type] += 1
        else:
            self._subscriptions[sensor_type] = 1
            self._start_sensor(sensor_type)

    @Slot(str)  # type: ignore[arg-type]
    def unsubscribe(self, sensor_type: str) -> None:
        """Unsubscribe from sensor updates."""
        if sensor_type in self._subscriptions:
            self._subscriptions[sensor_type] -= 1
            if self._subscriptions[sensor_type] <= 0:
                del self._subscriptions[sensor_type]
                self._stop_sensor(sensor_type)

    @Slot(str, result="QVariant")  # type: ignore[arg-type]
    def getLatestValue(self, sensor_type: str) -> Any:
        """Get the latest value for a sensor."""
        return self._latest_values.get(sensor_type)

    @Slot(result=list)  # type: ignore[arg-type]
    def getAvailableSensors(self) -> list[str]:
        """Get list of available sensor types."""
        return ["magnetometer", "gps", "barometer", "temperature", "humidity"]
```

### Backend Loader

```python
from __future__ import annotations

import importlib.util
from pathlib import Path
from typing import Callable

from PySide6.QtCore import QObject


class BridgeLoader:
    """Dynamically loads app backend modules."""

    def load_bridge(self, app_dir: Path, backend_path: str) -> QObject | None:
        """
        Load and instantiate an app's backend bridge.

        Args:
            app_dir: App's root directory
            backend_path: Relative path to backend module

        Returns:
            Instantiated bridge object or None
        """
        module_path = app_dir / backend_path
        if not module_path.exists():
            return None

        # Dynamically import the module
        spec = importlib.util.spec_from_file_location(
            f"app_backend_{app_dir.name}",
            module_path
        )
        if spec is None or spec.loader is None:
            return None

        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)

        # Look for create_bridge factory function
        if hasattr(module, "create_bridge"):
            return module.create_bridge()

        return None
```
